.data
prompt: .asciz "Input feature : "
forward: .asciz "\nRobot Action: FORWARD\n"
backward: .asciz "\nRobot Action: STOP\n"
stop: .asciz "\nRobot Action: BACKWARD\n"
invalid: .asciz "\nInvalid Shape!\n"
.text
.globl main
main:
    li a7,4
    la a0,prompt
    ecall
    li a7,5
    ecall
    mv t0,a0
    li t1,1
    beq t0,t1,triangle
    li t1,2
    beq t0,t1,square
    li t1,3
    beq t0,t1,circle
  
    li a7, 4
    la a0, invalid
    ecall
    j exit

circle:
    li a7, 4
    la a0, forward
    ecall
    j exit

square:
    li a7, 4
    la a0, stop
    ecall
    j exit

triangle:
    li a7, 4
    la a0, backward
    ecall

exit:
    li a7, 10
    ecall
